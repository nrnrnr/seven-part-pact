\newcommand\northsouth[5][3mm]{% [shift]{angle}{northtext}{southtext}{image}
  \node [white,anchor=north,yshift=-1*#1,rotate=#2] at (#5.north) {#3};
  \node [white,anchor=south,yshift=#1,rotate=#2] at (#5.south) {#4};
}

\newcommand\gamepiece[6]{% {basename}{width}{height}{diameter}{recto}{verso}
  \begingroup
  \def\basename{#1}%
  \pgfmathsetmacro\width{#2-2.5}%
  \pgfmathsetmacro\height{#3-2.5}%
  \def\diameter{#4}%
  \def\recto{#5}%
  \def\verso{#6}%
  \typeout{\basename: width=\width, height=\height, diameter=\diameter}%
  \begin{tikzpicture}[x=1mm,y=1mm]
     \large
     \node [rotate=180] (recto) at (0,0.5*\height) {\csname#1image\endcsname};
     \node (verso) at (0,-0.5*\height) {\expandafter\usebox\csname R\basename box\endcsname};
     \draw (-0.5*\width,-\height) rectangle ++(\width,2*\height);
     \draw [ultra thin] (-0.5*\width,0) -- ++(\width,0);
%     \draw [blue,thick] (0,0.5*\height) circle [radius=0.5*\diameter];
%     \draw [blue,thick] (0,-0.5*\height) circle [radius=0.5*\diameter];
     \northsouth[-2.5mm]{180}#5{recto}     
     \northsouth{0}#6{verso}     
  \end{tikzpicture}
  \endgroup
}
